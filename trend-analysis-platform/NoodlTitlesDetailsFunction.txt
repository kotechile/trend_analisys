// AFFILIATE PROGRAMS SECTION
// ================================================

// This section fetches affiliate programs automatically associated with the title
// during the Idea Burst publishing process. The affiliate_program_ids field
// contains an array of affiliate program UUIDs that match the title's keywords.
//
// ================================================

// Fetch affiliate programs for this title
let affiliatePrograms = [];
let monetizationData = {};

if (title_id && userID) {
    try {
        console.log(`🔍 Fetching affiliate opportunities for title: ${title_id}`);
        
        // Get affiliate opportunities data from the Titles table
        const { data: titleData, error: titleError } = await Noodl.Variables.supabase
            .from('Titles')
            .select('affiliate_opportunities, affiliate_program_ids, monetization_score, estimated_annual_revenue, revenue_breakdown, monetization_priority, monetization_analysis')
            .eq('id', title_id)
            .single();
        
        if (titleError) {
            console.warn("⚠️ Error fetching title data:", titleError);
        } else if (titleData) {
            // Try to get programs from affiliate_opportunities JSON field
            if (titleData.affiliate_opportunities) {
                try {
                    let oppData;
                    if (typeof titleData.affiliate_opportunities === 'string') {
                        oppData = JSON.parse(titleData.affiliate_opportunities);
                    } else {
                        oppData = titleData.affiliate_opportunities;
                    }
                    
                    console.log(`📋 Affiliate opportunities data:`, oppData);
                    
                    // Extract programs array from the JSON
                    if (oppData.programs && Array.isArray(oppData.programs)) {
                        affiliatePrograms = oppData.programs.map(program => ({
                            id: program.id,
                            program_name: program.name,
                            description: program.description,
                            commission_rate: program.commission_rate,
                            status: program.status,
                            network: program.network,
                            relevance_score: 75, // Default
                            estimated_monthly_revenue: 50, // Default
                            priority_level: 'medium' // Default
                        }));
                        console.log(`✅ Found ${affiliatePrograms.length} affiliate programs from opportunities`);
                    }
                } catch (parseError) {
                    console.warn("⚠️ Error parsing affiliate opportunities:", parseError);
                }
            }
            
            // Fallback: If no opportunities data, try to fetch from IDs
            if (affiliatePrograms.length === 0 && titleData.affiliate_program_ids) {
                try {
                    let affiliateIds;
                    if (typeof titleData.affiliate_program_ids === 'string') {
                        affiliateIds = JSON.parse(titleData.affiliate_program_ids);
                    } else if (Array.isArray(titleData.affiliate_program_ids)) {
                        affiliateIds = titleData.affiliate_program_ids;
                    }
                    
                    if (affiliateIds && affiliateIds.length > 0) {
                        console.log(`📋 Trying to fetch ${affiliateIds.length} programs from affiliate_offers table`);
                        const { data: affiliateData, error: affiliateError } = await Noodl.Variables.supabase
                            .from('affiliate_offers')
                            .select('id, offer_name, offer_description, commission_rate, status, network_name')
                            .in('id', affiliateIds);
                        
                        if (!affiliateError && affiliateData && affiliateData.length > 0) {
                            affiliatePrograms = affiliateData.map(program => ({
                                id: program.id,
                                program_name: program.offer_name,
                                description: program.offer_description,
                                commission_rate: program.commission_rate,
                                status: program.status,
                                network: program.network_name,
                                relevance_score: 75,
                                estimated_monthly_revenue: 50,
                                priority_level: 'medium'
                            }));
                            console.log(`✅ Found ${affiliatePrograms.length} affiliate programs from offers table`);
                        }
                    }
                } catch (fetchError) {
                    console.warn("⚠️ Error fetching affiliate offers:", fetchError);
                }
            }
        } else {
            console.log("ℹ️ No affiliate data found for this title");
        }
        
        // Extract monetization data from the title data we already fetched
        if (titleData) {
            monetizationData = {
                monetization_score: titleData.monetization_score || 0,
                estimated_annual_revenue: titleData.estimated_annual_revenue || 0,
                affiliate_opportunities: titleData.affiliate_opportunities || {},
                revenue_breakdown: titleData.revenue_breakdown || {},
                monetization_priority: titleData.monetization_priority || 'low',
                monetization_analysis: titleData.monetization_analysis || {}
            };
            console.log("💰 Monetization data loaded:", monetizationData);
        }
        
    } catch (error) {
        console.warn("⚠️ Error in affiliate program fetch:", error);
    }
}
